# === step2 파라미터 템플릿 — docker cc → docker lms 직결 ===
# 사용: cp .env.org .env && vi .env

# === 모델 ===
# ⚠️ 보안망 정책: 중국계(Qwen·GLM 등) 금지. 선택 우선은 Google(gemma) 계열.
#   16GB GPU 검증 머신 : gemma-4-e2b-it        (4.41GB — 단일/다중 백엔드 모두 여유)
#   A6000 반입 서버    : gemma-4-31b-it-qat     (17.6GB — lms:small 이미지에 함께 내장)
#   두 모델 모두 lms:small 이미지에 내장되어 있어 이미지 교체 없이 모델 키만 바꾸면 된다.
#
# ⚠️ **허브 키가 아니라 로컬 키를 쓸 것** (2026-07-20 step2 실측).
#   매니페스트 없이 gguf 만 반입한 모델은 로컬 키가 허브 키와 다르다.
#   허브 키(google/gemma-4-e2b)를 쓰면 'lms load' 가 1회 실패한 뒤 폴백 경로로 겨우 로드된다
#   — 동작은 하지만 기동이 느려지고 로그가 실패로 오염되므로 반입본에 남기지 않는다.
#   확인: docker exec lms sh -lc 'PATH=$HOME/.lmstudio/bin:$PATH lms ls'
LMS_MODEL=gemma-4-e2b-it
LMS_PORT=1234
# 서버 헬스 폴링 최대 횟수 (×2초). 대형 모델 초기화가 느리면 늘릴 것.
LMS_HEALTH_TRIES=30
# 컨텍스트 토큰 상한. Claude Code 시스템+도구 페이로드가 크므로 32768 미만이면 동작 불가.
LMS_CONTEXT_LENGTH=32768
# GPU offload 비율 ("max"/"off"/0~1). 빈값=자동.
#   e2b(3.4GB)는 max 로 전량 탑재 가능. 31b(17.6GB)를 16GB GPU 에서 쓸 때만 0.6 등 부분 offload.
#   ⚠️ VRAM 83% 안전선: KV 는 로드 시 선할당되지만 prefill 연산 버퍼는 추론 시점에 추가로 필요.
#      로드 직후 97% 구성이 장문 프롬프트에서 CUDA OOM 크래시 (실측).
LMS_GPU=max
# 동시 예측 슬롯 — llmster 는 컨텍스트를 슬롯 수로 "분할"(기본 4 → 슬롯당 ctx/4).
#   Claude Code 시스템 프롬프트가 슬롯을 초과하면 500 (n_keep >= n_ctx). 반드시 1.
LMS_PARALLEL=1
# 1 = 기동 시 'lms get'(허브 접속) 완전 생략 — 폐쇄망 필수.
#   blackhole 방화벽에서 'lms get' 은 실패가 아니라 수 분 hang 하여 모델 로드까지 도달 못 함.
LMS_SKIP_GET=1
LMS_GET_TIMEOUT=60
# 모델 저장소: 빈값=named volume, 절대경로=bind mount.
#   lms:small 이미지에 모델이 내장되어 있으므로 빈 볼륨을 덮어씌우지 않도록 주의
#   (빈 named volume 을 /home/lms/.lmstudio/models 에 마운트하면 이미지 내장 모델이 가려진다).
#   step2 기본값은 '마운트 없음' — run.sh 가 LMS_MODEL_MOUNT 빈값이면 마운트를 생략한다.
LMS_MODEL_MOUNT=

# === Claude Code ===
CLAUDE_MAX_OUTPUT_TOKENS=8192
CLAUDE_DIET=1
API_TIMEOUT_MS=600000
MOUNT_CODE_DIR=

# === 컨테이너 / 네트워크 ===
LMS_CONTAINER_NAME=lms
CLAUDE_CONTAINER_NAME=cc-step2
LMS_NETWORK_NAME=lms-step2
LMS_IMAGE=lms:small
CLAUDE_IMAGE=claude:latest
TZ=Asia/Seoul

# === GPU ===
# 1 = --gpus all (NVIDIA 드라이버 + nvidia-container-toolkit 필요), 0 = CPU
USE_GPU=1
# lms 포트를 호스트에 노출할지 (선택, 디버깅용). 빈값이면 노출 안 함.
#   step2 판정은 'cc 컨테이너에서의 접근'이 기준이므로 호스트 노출은 필수가 아니다.
LMS_PUBLISH_PORT=
