# === 6.lms_MultiLLM_run 파라미터 템플릿 (air-gap 검증본) ===
# 사용: cp .env.org .env  (이 폴더의 .env 가 이미 검증된 working copy — 그대로 써도 됨)

# === 모델 ===
# 기반입 lms:latest 이미지에 내장된 유일한 LLM. 변경하려면 모델을 별도 반입해야 함.
LMS_MODEL=google/gemma-4-31b-qat
LMS_PORT=1234
# 서버 헬스 폴링 최대 횟수 (×2초). 대형 모델 초기화가 느리면 늘릴 것.
LMS_HEALTH_TRIES=30
# 컨텍스트 토큰 상한. Claude Code 는 시스템+도구 페이로드 ~16k → 32768 미만이면 동작 불가.
#   (CLAUDE_DIET=1 이면 ~3k 로 줄지만, 대화가 길어지므로 32768 유지 권장)
#   상향 가능하나 32k 유지 권장 — 근거는 info_modelTuning.md §3.
#   (구 주석의 '측정: 64k≈36GB, 128k≈54GB' 는 출처 없는 추정이었고 자체 모순이라 삭제함.
#    gemma-4 는 5:1 sliding-window 구조라 KV 가 컨텍스트에 거의 비례하지 않는다 —
#    32k→128k 증분이 fp16 기준 수 GB 수준. 즉 128k 를 막는 것은 VRAM 이 아니라
#    prefill 지연이며, 31b 의 A6000 prefill 은 미측정이다. 올리려면 API_TIMEOUT_MS 동반 상향.)
# ⚠️ VRAM 83% 안전선(fg1 실측): KV 는 로드 시 선할당되지만 prefill 연산 버퍼는 추론 시점에
#   추가로 필요 — 로드 후 VRAM 97% 구성이 41k 프롬프트에서 CUDA OOM 크래시.
#   짧은 프롬프트 통과만 보고 판단하지 말고 로드 후 사용률 ≤83% 를 확인할 것.
LMS_CONTEXT_LENGTH=32768
# GPU offload: A6000(48GB, air-gap 서버) 은 "max" 권장 — 31b(18.85GB)+32k KV 전체 탑재 가능.
#   16GB 급(검증 머신)은 빈값(자동) 또는 0.x 부분 offload (max 는 CUDA OOM).
LMS_GPU=max
# 동시 예측 슬롯 — llmster 는 컨텍스트를 슬롯 수로 "분할"(기본 4 → 슬롯당 ctx/4).
#   Claude Code 시스템 프롬프트(~16k)가 슬롯 초과 시 500 (n_keep >= n_ctx). 반드시 1.
LMS_PARALLEL=1
# 'lms get'(LM Studio 허브 접속) 생략 — 폐쇄망은 1 고정 권장.
#   0 으로 두면 blackhole 방화벽(패킷 drop)에서 '실패'가 아니라 수 분간 hang 한다.
LMS_SKIP_GET=1
# LMS_SKIP_GET=0 일 때만 의미 있음. 'lms get' 상한(초).
LMS_GET_TIMEOUT=60
# 모델 저장소: 빈값=named volume(lms-models). lms:latest 이미지에 gemma 가 내장되어 있으므로
#   볼륨 마운트 없이도 동작 — 기본값 유지 권장.
LMS_MODEL_MOUNT=

# === 게이트웨이 (nginx L7 분산, 단일 외부 주소) ===
GATEWAY_PORT=8080

# === 백엔드 수 ===
# 6.판에서는 실제 기동 개수 (lms-1..N). GPU/RAM 용량 검토 후 증설.
# A6000(48GB) 산정(추정 — 반입 후 실측 필수, VRAM 83%≈40GB 안전선):
#   31b@32k ×1 ≈ 25~33GB(52~69%) ✓  /  같은 GPU 에 ×2 ≈ 50~66GB ⚠️ 48GB 초과
#   (가중치 18.85 + KV 3~9(층 수 미확인 대역) + 연산버퍼 3~5. 상세: info_modelTuning.md §3)
#   → GPU 당 1백엔드가 원칙. start.sh 가 nvidia-smi 로 GPU 수를 세어 백엔드를
#     GPU 0,1,2… 에 1:1 로 핀하고(--gpus "device=K"), 백엔드 > GPU 면 OOM 경로이므로
#     기동을 중단한다. 즉 GPU 1장 머신이면 이 값을 1 로 낮춰야 기동된다.
#     (VRAM 이 확실히 감당될 때만 LMS_ALLOW_GPU_OVERSUBSCRIBE=1 로 강행 가능)
LMS_BACKEND_COUNT=2

# === Claude ↔ 게이트웨이 ===
# 로컬 추론 지연 대비 타임아웃(ms). 기본 10분.
API_TIMEOUT_MS=600000
# 출력 토큰 상한 — 32k 컨텍스트에 input+output 이 함께 들어가야 함.
#   Claude Code 기본 max_tokens(≈32000)가 그대로 가면 컨텍스트 초과 → 500.
CLAUDE_MAX_OUTPUT_TOKENS=8192
# 프롬프트 다이어트: 1=도구 4개(Read/Bash/Edit/Write)만 노출 — 프롬프트 19.4k→3.0k(-84%),
#   왕복 3.7~6.8배 단축 실측(fg1). 0=전체 도구. SSOT: DeviceManagement fg1/lms/diet_settings.json
CLAUDE_DIET=1
# 호스트 코드 폴더 마운트 (선택) — 지정 시 /home/ubuntu/code 로 마운트
MOUNT_CODE_DIR=

# === 컨테이너/네트워크 ===
GATEWAY_CONTAINER_NAME=gateway
CLAUDE_CONTAINER_NAME=claude
LMS_NETWORK_NAME=lms

# === 이미지 태그 (이 폴더의 tar 를 docker load 한 결과와 일치) ===
LMS_IMAGE=lms:latest
GATEWAY_IMAGE=lms-gateway:latest
CLAUDE_IMAGE=claude:latest

# === 사용자 UID/GID · 타임존 ===
USER_UID=1000
USER_GID=1000
TZ=Asia/Seoul

# === GPU ===
# 1 = --gpus all (호스트에 NVIDIA 드라이버 + nvidia-container-toolkit 필요), 0 = CPU
USE_GPU=1
