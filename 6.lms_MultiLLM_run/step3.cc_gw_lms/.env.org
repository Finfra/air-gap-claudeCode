# === step3 파라미터 템플릿 — docker cc → docker gateway → docker lms ===
# 사용: cp .env.org .env && vi .env

# === 모델 ===
# ⚠️ 보안망 정책: 중국계(Qwen·GLM 등) 금지. 선택 우선은 Google(gemma) 계열.
#   16GB GPU 검증 머신 : gemma-4-e2b-it        (4.41GB)
#   A6000 반입 서버    : gemma-4-31b-it-qat     (17.6GB — 같은 이미지에 내장)
# ⚠️ 허브 키가 아니라 **로컬 키**를 쓸 것 (step2 실측 — 허브 키는 1회 로드 실패 후 폴백).
#   확인: docker exec lms-1 sh -lc 'PATH=$HOME/.lmstudio/bin:$PATH lms ls'
LMS_MODEL=gemma-4-e2b-it
LMS_PORT=1234
LMS_HEALTH_TRIES=30
LMS_CONTEXT_LENGTH=32768
LMS_GPU=max
# 슬롯 분할 방지 — Claude Code 는 1 필수 (ctx/슬롯수 = 슬롯당 ctx)
LMS_PARALLEL=1
# 폐쇄망 필수 — 허브 확인이 blackhole 방화벽에서 hang
LMS_SKIP_GET=1
LMS_GET_TIMEOUT=60
# 빈값 = 마운트 없음(이미지 내장 모델 사용). 빈 named volume 을 물리면 내장 모델이 가려진다.
LMS_MODEL_MOUNT=

# === 백엔드 수 ===
# step3 은 게이트웨이 계층만 검증하는 단계이므로 **1 고정**.
#   다중화(2개)는 step4 소관 — 여기서 늘리면 변수 단일 증분 원칙이 깨진다.
LMS_BACKEND_COUNT=1

# === 게이트웨이 (nginx L7 패스스루) ===
# 호스트에 노출할 포트
GATEWAY_PORT=8080
# 바인딩 주소. 기본은 호스트 내부 전용.
#   LAN(Windows 클라이언트) 노출은 step5 소관 — 여기서는 127.0.0.1 유지.
GATEWAY_BIND_HOST=127.0.0.1
# 게이트웨이 컨테이너 내부 리슨 포트 (nginx.conf.template 의 listen 과 일치해야 함)
GATEWAY_INTERNAL_PORT=8080

# === Claude Code ===
CLAUDE_MAX_OUTPUT_TOKENS=8192
CLAUDE_DIET=1
API_TIMEOUT_MS=600000
MOUNT_CODE_DIR=

# === 컨테이너 / 네트워크 ===
GATEWAY_CONTAINER_NAME=gateway
CLAUDE_CONTAINER_NAME=cc-step3
LMS_NETWORK_NAME=lms-step3
LMS_IMAGE=lms:small
GATEWAY_IMAGE=lms-gateway:latest
CLAUDE_IMAGE=claude:latest
TZ=Asia/Seoul

# === GPU ===
USE_GPU=1
