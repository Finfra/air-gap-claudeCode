# 8.vllm_MultiLLM 환경값 — cp .env.org .env 후 수정
# 기본 프리셋: 16GB GPU 1장에 소형 모델 복제본 2개 (멀티 백엔드 분산 검증용)
#   27B 단일 백엔드 프리셋은 qwen38-27b.env.org (cp qwen38-27b.env.org .env)

# === 모델 (전 복제본 공통) ===
# HF repo id 또는 GGUF(<repo>:<quant>). 폐쇄망이면 HF_CACHE_DIR 에 미리 받아 두고 VLLM_OFFLINE=1
#   Qwen3-4B-Instruct-2507-FP8: 4.4GB, non-thinking, tool call=hermes, ctx 262k
VLLM_MODEL=Qwen/Qwen3-4B-Instruct-2507-FP8
# API 상의 모델 이름 — claude 의 ANTHROPIC_MODEL 로도 그대로 전달
SERVED_MODEL_NAME=qwen3-4b
# 모델 계열별 파서: Qwen3 Instruct → hermes / Qwen3.5·3.8·Coder → qwen3_coder (빈값=tool call 끔)
TOOL_CALL_PARSER=hermes
# thinking 모델이면 qwen3 (사고 과정을 reasoning 으로 분리), non-thinking 이면 빈값
REASONING_PARSER=

# === 복제본 ===
# 백엔드 수. 'up --scale vllm=N' 을 주면 그 값이 우선
VLLM_REPLICAS=2
# 복제본 1개가 쓸 GPU 메모리 비율 — GPU 1장을 N개가 나누면 합이 0.9 이하가 되게 (2개 → 0.45)
GPU_MEMORY_UTILIZATION=0.45
# 복제본을 하나씩 순서대로 로드 (1 권장 — 동시 로드는 메모리 프로파일링이 서로 오염됨)
VLLM_STARTUP_LOCK=1

# === TurboQuant KV cache (vLLM 0.30 upstream 내장) ===
#   auto                 모델 기본 dtype (압축 없음)
#   turboquant_k8v4      K=FP8,  V=4bit  (품질 우선)
#   turboquant_4bit_nc   K=4bit, V=4bit  ← 기본값
#   turboquant_k3v4_nc   K=3bit, V=4bit
#   turboquant_3bit_nc   K=3bit, V=3bit  (최대 압축)
KV_CACHE_DTYPE=turboquant_4bit_nc

# === 컨텍스트·배치 ===
# Claude Code 는 시스템 프롬프트만 3k(다이어트)~19k(전체) 토큰 → 32k 이상 권장
MAX_MODEL_LEN=32768
# 복제본당 동시 시퀀스 수
MAX_NUM_SEQS=4
# prefill 1회 처리 토큰 수 — 클수록 빠르나 활성화 메모리가 커짐
MAX_NUM_BATCHED_TOKENS=2048
# 가중치가 VRAM 을 넘으면 초과분을 CPU RAM 으로 (0 = 미사용, 늘릴수록 느려짐)
CPU_OFFLOAD_GB=0
# 추가 vllm serve 인자 (공백 분리) ex) --enforce-eager (CUDA graph 생략 → 기동 빠름·VRAM 절약·생성 느림)
VLLM_EXTRA_ARGS=

# === 저장소 ===
# 모델(HF) 캐시: 호스트 경로면 공유(재다운로드 방지), 빈값이면 named volume hf-cache
HF_CACHE_DIR=~/.cache/huggingface
# torch.compile 캐시: 호스트 경로 또는 빈값(named volume vllm-cache)
VLLM_CACHE_DIR=~/.cache/vllm
# 1 = HF 접속 차단 (모델이 캐시에 있을 때 — 폐쇄망 필수)
VLLM_OFFLINE=0
# gated/비공개 모델용 (공개 모델이면 비워둠)
HF_TOKEN=

# === 게이트웨이 ===
# 외부 노출 포트(호스트). 컨테이너 내부는 8080 고정
GATEWAY_PORT=8080

# === Claude ↔ 게이트웨이 ===
API_TIMEOUT_MS=600000
# Claude Code 출력 토큰 상한 — input+output 이 MAX_MODEL_LEN 안에 들어가야 함
CLAUDE_MAX_OUTPUT_TOKENS=8192
# 추론 effort (low·medium·xhigh) — Qwen3.8 템플릿은 high 를 거부한다(Claude Code 기본값이 high)
CLAUDE_EFFORT_LEVEL=medium
# 프롬프트 다이어트: 1=핵심 도구만(프롬프트 -84%), 0=전체 (WebSearch·WebFetch 는 항상 deny)
CLAUDE_DIET=1
# 공유 작업 폴더 → claude /home/ubuntu/df
DF_DIR=~/df
# 호스트 코드 폴더 마운트 (선택) — docker-compose.code.yml override 사용 시
MOUNT_CODE_DIR=

# === 컨테이너·네트워크 (다중 인스턴스 동시 실행 시 충돌 회피) ===
COMPOSE_PROJECT_NAME=air_gap_claude_code_vllm_multi
GATEWAY_CONTAINER_NAME=vllm-gateway
CLAUDE_CONTAINER_NAME=vllm-claude
VLLM_NETWORK_NAME=vllm

# === 사용자 UID/GID (claude 컨테이너 — 호스트 파일 권한 일치) ===
USER_UID=1000
USER_GID=1000

# 구형 드라이버(525) 우회: 이미지의 cuda>=12.9 요구 검사 해제. 드라이버 575+ 면 0
NVIDIA_DISABLE_REQUIRE=1

TZ=Asia/Seoul
