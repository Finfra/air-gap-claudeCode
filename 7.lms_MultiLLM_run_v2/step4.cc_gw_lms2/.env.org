# === step4 파라미터 템플릿 — docker cc → gateway → lms-1, lms-2 ===
# 사용: cp .env.org .env && vi .env

# === 모델 ===
# ⚠️ 보안망 정책: 중국계(Qwen·GLM 등) 금지. 선택 우선은 Google(gemma) 계열.
# ⚠️ 허브 키가 아니라 **로컬 키**를 쓸 것 (step2 실측).
#   반입본(lms:31b-cuda11fix)에 내장된 유일한 LLM 은 google/gemma-4-31b-qat (17.6GB) 하나다.
#   `lms ls` 로 확인한 로컬 키가 그대로 이 값이며, 클라이언트의 ANTHROPIC_MODEL 도 동일해야 함.
#   ex) A6000(48GB) 전량 offload → LMS_GPU=max
#       16GB GPU 에서 억지로 띄우려면 부분 offload 필수(LMS_GPU=0.6, 매우 느림 — 검증용만)
LMS_MODEL=google/gemma-4-31b-qat
LMS_PORT=1234
LMS_HEALTH_TRIES=30
LMS_CONTEXT_LENGTH=32768
LMS_GPU=max
LMS_PARALLEL=1
LMS_SKIP_GET=1
LMS_GET_TIMEOUT=60
LMS_MODEL_MOUNT=

# === 백엔드 수 (이 단계의 핵심 변수) ===
# ⚠️ 31B 반입본 기준: A6000(48GB) + 31b(17.6GB) ×2 ≈ 44GB (92%) → 83% 안전선 초과.
#   2 로 올리기 전에 nvidia-smi 로 실측 VRAM 을 확인할 것. 기본은 1(단일 백엔드, 안전).
LMS_BACKEND_COUNT=1

# GPU 배치: 백엔드 1개당 GPU 1장 전용 할당(--gpus "device=K").
#   '--gpus all' 을 N개 백엔드에 똑같이 주면 전원이 같은 GPU0 에 적재된다.
#   백엔드 수 > GPU 수이면 run.sh 가 기본적으로 정지한다 — 초과 구독은 OOM 경로이기 때문.
#   VRAM 이 (모델+KV)×백엔드 를 확실히 감당할 때만 아래를 1 로 켜서 강행할 것.
#   ex) fg1(GPU 1장 16GB) + e2b 4.41GB ×2 ≈ 8GB (49%) → 안전, 승인하고 진행
#       A6000(48GB) + 31b ×2 ≈ 44GB (92%) → 83% 안전선 초과, 승인 금지
LMS_ALLOW_GPU_OVERSUBSCRIBE=1

# === 게이트웨이 ===
GATEWAY_PORT=8080
GATEWAY_BIND_HOST=127.0.0.1
GATEWAY_INTERNAL_PORT=8080

# === Claude Code ===
CLAUDE_MAX_OUTPUT_TOKENS=8192
CLAUDE_DIET=1
API_TIMEOUT_MS=600000
MOUNT_CODE_DIR=

# === 컨테이너 / 네트워크 ===
GATEWAY_CONTAINER_NAME=gateway
CLAUDE_CONTAINER_NAME=cc-step4
LMS_NETWORK_NAME=lms-step4
# v2: CUDA 11 런타임(libcudart.so.11.0) 노출 오버레이 적용본. cuda11fix/ 참조.
#   air-gap 반입은 _img/ 로 export/load 한 이 태그를 그대로 사용(cuda11fix 반영본).
#   lms:31b-cuda11fix = 31B 전용 슬림본(22.2GB). cuda11fix/build_slim31b.sh 산출물.
#   E2B 도 필요하면 lms:small-cuda11fix(29.1GB, 두 모델 동봉)로 바꿔 export/load 할 것.
LMS_IMAGE=lms:31b-cuda11fix
GATEWAY_IMAGE=lms-gateway:latest
CLAUDE_IMAGE=claude:latest
TZ=Asia/Seoul

# === GPU ===
USE_GPU=1
# v2: CUDA 런타임(libcudart.so.11.0) 프리플라이트. 1이면 CUDA 미해결 시 CPU 폴백 없이 중단.
#   6.lms_MultiLLM_run 반입 실패 주 원인(libcudart.so.11.0 부재)을 조용히 넘기지 않기 위함.
#   prj55#Issue9(CUDA 11 런타임 설치) 완료 후에만 GPU 로 통과됨. CPU 강행은 0.
LMS_REQUIRE_CUDA=1
